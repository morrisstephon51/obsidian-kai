"""
Edge-case test suite for CostTracker.
Assumes interface:
    CostTracker(budget: float, currency: str = "USD")
    .add(amount: float, category: str, description: str = "", date: datetime | None = None) -> Transaction
    .refund(transaction_id: str, amount: float | None = None) -> Transaction
    .total(category: str | None = None) -> float
    .remaining() -> float
    .transactions(start: datetime | None = None, end: datetime | None = None) -> list[Transaction]
    .summary() -> dict[str, float]
"""

import pytest
from datetime import datetime, timezone, timedelta
from decimal import Decimal
from unittest.mock import patch

from cost_tracker import CostTracker, Transaction, InsufficientBudgetError, InvalidAmountError


# ──────────────────────────────────────────────────────────────
# Fixtures
# ──────────────────────────────────────────────────────────────

@pytest.fixture
def tracker():
    return CostTracker(budget=1_000.00)


@pytest.fixture
def empty_tracker():
    return CostTracker(budget=0.00)


# ──────────────────────────────────────────────────────────────
# Scenario 1 — Zero-amount transaction
# ──────────────────────────────────────────────────────────────

def test_zero_amount_raises(tracker):
    """A $0.00 charge carries no semantic value and should be rejected."""
    with pytest.raises(InvalidAmountError, match="must be positive"):
        tracker.add(0.00, "compute")


# ──────────────────────────────────────────────────────────────
# Scenario 2 — Negative amount (attempted direct negative charge)
# ──────────────────────────────────────────────────────────────

def test_negative_amount_raises(tracker):
    """Negative amounts on add() are not refunds; refund() is the correct API."""
    with pytest.raises(InvalidAmountError):
        tracker.add(-50.00, "storage")


# ──────────────────────────────────────────────────────────────
# Scenario 3 — Floating-point precision accumulation
# ──────────────────────────────────────────────────────────────

def test_floating_point_accumulation_does_not_drift(tracker):
    """
    Repeatedly adding $0.10 ten times must equal $1.00, not $0.9999…
    CostTracker must use Decimal or round internally.
    """
    for _ in range(10):
        tracker.add(0.10, "api-calls")

    assert tracker.total(category="api-calls") == pytest.approx(1.00, abs=1e-9)


# ──────────────────────────────────────────────────────────────
# Scenario 4 — Exact budget boundary (spend == budget)
# ──────────────────────────────────────────────────────────────

def test_spend_exactly_at_budget_allowed(tracker):
    """Spending the last cent of the budget is valid; remaining must be 0."""
    tracker.add(1_000.00, "infra")
    assert tracker.remaining() == pytest.approx(0.00, abs=1e-9)


# ──────────────────────────────────────────────────────────────
# Scenario 5 — Exceeding budget
# ──────────────────────────────────────────────────────────────

def test_exceeding_budget_raises(tracker):
    """A single charge that pushes total over budget must raise."""
    tracker.add(999.99, "infra")
    with pytest.raises(InsufficientBudgetError):
        tracker.add(0.02, "infra")   # $0.02 would exceed by $0.01


# ──────────────────────────────────────────────────────────────
# Scenario 6 — Partial refund restores budget correctly
# ──────────────────────────────────────────────────────────────

def test_partial_refund_restores_budget(tracker):
    tx = tracker.add(200.00, "ml-training")
    tracker.refund(tx.id, amount=75.00)

    assert tracker.total() == pytest.approx(125.00, abs=1e-9)
    assert tracker.remaining() == pytest.approx(875.00, abs=1e-9)


# ──────────────────────────────────────────────────────────────
# Scenario 7 — Full refund on zero-budget tracker
# ──────────────────────────────────────────────────────────────

def test_full_refund_on_zero_budget_tracker():
    """
    Tracker with $0 budget: after adding a charge (budget check skipped

    Alternatively, a $0 budget tracker could be unlimited — test documents

    t = CostTracker(budget=0.00)          # 0 == unlimited in this design

    t.refund(tx.id)                        # full refund, no amount arg

def test_refund_exceeding_original_amount_raises(tracker):

# Scenario 9 — Double refund of the same transaction

def test_double_full_refund_raises(tracker):
    tx = tracker.add(100.00, "cdn")
    tracker.refund(tx.id)
    with pytest.raises((InvalidAmountError, LookupError)):
        tracker.refund(tx.id)       # already fully refunded


# ──────────────────────────────────────────────────────────────
# Scenario 10 — Date-range filtering excludes boundary extremes
# ──────────────────────────────────────────────────────────────

def test_date_range_filtering_is_inclusive_exclusive(tracker):
    """
    .transactions(start, end) should be [start, end) — inclusive start,
    exclusive end — matching standard Python range semantics.
    """
    t0 = datetime(2025, 1,  1, tzinfo=timezone.utc)
    t1 = datetime(2025, 1, 15, tzinfo=timezone.utc)
    t2 = datetime(2025, 2,  1, tzinfo=timezone.utc)

    tracker.add(10.00, "a", date=t0)
    tracker.add(20.00, "b", date=t1)
    tracker.add(30.00, "c", date=t2)

    results = tracker.transactions(start=t0, end=t2)
    amounts = {tx.amount for tx in results}

    assert 10.00 in amounts
    assert 20.00 in amounts
    assert 30.00 not in amounts      # t2 is excluded (end is exclusive)

def test_summary_reflects_net_amounts_after_refunds(tracker):

# Scenario 15 — Concurrent-safe idempotency key prevents duplicate charges
# ──────────────────────────────────────────────────────────────

def test_duplicate_idempotency_key_is_deduplicated(tracker):
    """
    If the same idempotency_key is supplied twice, the second call must

    assert tracker.total() == pytest.approx(50.00, abs=1e-9)   # charged once
